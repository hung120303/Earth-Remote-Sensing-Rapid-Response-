// S2ImageVisualization

// Pick a specific point source to extract an image from via coordinates
var longitude = -90.64; //W-E
var latitude = 34.14; //N-S
var startDate = '2023-01-19';
var endDate = '2023-02-06';
var point = ee.Geometry.Point(longitude, latitude);

// Define a function for obtaining a bounding box around each feature
var getBoundingBox = function(feature) {
  var buffer = feature.buffer({
    distance: 2560,
  }); // radius of roughly 5120 meters (5.12km)
  var box = buffer.bounds();
  return box;
};

var boundingBoxPoint = getBoundingBox(ee.Feature(point));
print (boundingBoxPoint);

// Convert feature geometry to region of interest
var roi = boundingBoxPoint.geometry();
print(roi);

// Import Sentinel2 Imagery (harmonized)
var S2_Harmonized = ee.ImageCollection('COPERNICUS/S2_HARMONIZED')
.filterBounds(roi)
.filterMetadata('CLOUD_COVERAGE_ASSESSMENT', 'LESS_THAN', 5)
.filterDate(startDate, endDate)
.select(['B2', 'B3', 'B4', 'B11', 'B12']);

print(S2_Harmonized);

// There may be overlapping images: get the first one
var S2Image = S2_Harmonized.first();
print(S2Image);

// resample bands B2, B3, B4 (10m per pixel) to match B11, B12 (20m per pixel)
var S2ImageProjection = S2Image.select('B12').projection();
var S2ImageBands = S2Image.select('B2','B3','B4','B11','B12');
var S2ImageResample = S2ImageBands.reproject({
  crs: 'EPSG:4326',
  scale: 20
});
print(S2ImageResample);

// Normalize image (Sentinel 2 imagery has values of 0-10000, 
// so divide by 10000 to get values between 0-1)
var S2ImageRescale = S2ImageResample.divide(1); // set to 1 temporarily
var clip = S2ImageRescale.clip(roi);
var clipS2Image = ee.Image(clip.copyProperties(S2ImageResample));
print(clipS2Image);

var visParam = {bands: ['B4', 'B3', 'B2'], min: 0, max: 0.3};
Map.centerObject(boundingBoxPoint, 10);
Map.addLayer(boundingBoxPoint, {color: '#FECA1E', fillColor:'#4c4cff'}, 'Bounding Box');
Map.addLayer(clipS2Image, visParam, 'Sentinel 2022');

// Get base image projection info and id for use in export
var projection = S2ImageResample.projection().getInfo();
var S2ImageID = S2ImageResample.id();
var id = S2ImageID.getInfo();

// Export Image
Export.image.toDrive({
  image: clipS2Image,
  description: id,
  folder: "S2_Images",
  dimensions: '256x256',
  crs: projection.crs,
  crsTransform: projection.transform,
  region: roi
});
